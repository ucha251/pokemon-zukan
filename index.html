const fs = require('fs'); const TYPE_MAP = { normal: "ノーマル", fire: "ほのお", water: "みず", electric: "でんき", grass: "くさ", ice: "こおり", fighting: "かくとう", poison: "どく", ground: "じめん", flying: "ひこう", psychic: "エスパー", bug: "むし", rock: "いわ", ghost: "ゴースト", dragon: "ドラゴン", dark: "あく", steel: "はがね" }; const PHYSICAL_TYPES = ["ノーマル", "かくとう", "どく", "じめん", "ひこう", "むし", "いわ", "ゴースト", "はがね"]; const moveCache = {}; const movesDB = {}; const pokedex = []; async function getJapaneseName(url) { const res = await fetch(url); const data = await res.json(); const ja = data.names.find(n => n.language.name === "ja-Hrkt" || n.language.name === "ja"); return ja ? ja.name : data.name; } async function getMoveData(moveUrl) { if (moveCache[moveUrl]) return moveCache[moveUrl]; const res = await fetch(moveUrl); const data = await res.json(); const jaName = await getJapaneseName(moveUrl); const typeJa = TYPE_MAP[data.type.name] || "ノーマル"; let category = "変化"; if (data.damage_class.name !== "status") { category = PHYSICAL_TYPES.includes(typeJa) ? "物理" : "特殊"; } let power = data.power || "-"; if (["10まんボルト", "かえんほうしゃ", "れいとうビーム", "なみのり"].includes(jaName)) power = 95; if (jaName === "だいもんじ") power = 120; movesDB[jaName] = { type: typeJa, category: category, power: power, accuracy: data.accuracy !== null ? data.accuracy : "-", pp: data.pp || "-" }; moveCache[moveUrl] = jaName; return jaName; } async function parseCompleteEvolutionTree(node, nameMap) { const currentEng = node.species.name; const currentJa = nameMap[currentEng]; const results = []; for (const next of node.evolves_to) { const nextEng = next.species.name; const nextJa = nameMap[nextEng]; const d = next.evolution_details[0] || {}; let method = "特殊"; let cond = "条件あり"; if (d.min_level) { method = "レベルアップ"; cond = `Lv.${d.min_level}`; } else if (d.item) { method = "進化の石・道具"; cond = d.item.name; } else if (d.trigger && d.trigger.name === "trade") { method = "通信交換"; cond = "通信"; } else if (d.min_happiness) { method = "なつき"; cond = "なつき度MAX"; } results.push({ from: currentJa, to: nextJa, method: method, condition: cond }); const subResults = await parseCompleteEvolutionTree(next, nameMap); results.push(...subResults); } return results; } async function main() { console.log("No.1〜251のデータを取得中..."); const speciesList = []; for (let i = 1; i <= 251; i++) { const [pokeRes, specRes] = await Promise.all([ fetch(`https://pokeapi.co/api/v2/pokemon/${i}`), fetch(`https://pokeapi.co/api/v2/pokemon-species/${i}`) ]); const poke = await pokeRes.json(); const spec = await specRes.json(); const jaName = spec.names.find(n => n.language.name === "ja-Hrkt" || n.language.name === "ja").name; speciesList.push({ i, poke, spec, jaName, englishName: spec.name }); } const nameMap = {}; speciesList.forEach(s => { nameMap[s.englishName] = s.jaName; }); const chainCache = {}; for (const item of speciesList) { const chainUrl = item.spec.evolution_chain.url; if (!chainCache[chainUrl]) { const evoRes = await fetch(chainUrl); const evoData = await evoRes.json(); chainCache[chainUrl] = await parseCompleteEvolutionTree(evoData.chain, nameMap); } } for (const item of speciesList) { const { i, poke, spec, jaName } = item; const types = poke.types.map(t => TYPE_MAP[t.type.name]); const chainUrl = spec.evolution_chain.url; const evolution_family = chainCache[chainUrl] || []; const levelUp = []; const tmSet = new Set(); const hmSet = new Set(); const HM_NAMES = ["いあいぎり", "そらをとぶ", "なみのり", "かいりき", "フラッシュ", "うずしお", "たきのぼり"]; for (const m of poke.moves) { const gsDetail = m.version_group_details.find(v => v.version_group.name === "gold-silver"); const crDetail = m.version_group_details.find(v => v.version_group.name === "crystal"); const detail = gsDetail || crDetail; if (!detail) continue; const moveName = await getMoveData(m.move.url); if (detail.move_learn_method.name === "level-up") { const ver = (!gsDetail && crDetail) ? "クリスタル" : "共通"; levelUp.push({ level: detail.level_learned_at, move: moveName, version: ver }); } else if (detail.move_learn_method.name === "machine") { if (HM_NAMES.includes(moveName)) hmSet.add(moveName); else tmSet.add(moveName); } } levelUp.sort((a, b) => a.level - b.level); pokedex.push({ no: i, name: jaName, types, evolution_family, level_up_moves: levelUp, tm_moves: Array.from(tmSet), hm_moves: Array.from(hmSet), extra: {} }); if (i % 10 === 0 || i === 251) console.log(`${i} / 251 匹完了...`); } fs.writeFileSync('moves.json', JSON.stringify(movesDB, null, 2)); fs.writeFileSync('pokedex.json', JSON.stringify(pokedex, null, 2)); console.log("🎉 完了！図鑑データが更新されました！"); } main();
